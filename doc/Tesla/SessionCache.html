Overview

Persistent, file-backed store for signed-command session state. It uses an exclusive lock and atomic replacement so multiple processes can reserve monotonically increasing counters without corrupting the cached session.

See Also:

def load_unlocked
  return nil unless File.file?(@path)

  begin
    data =
      JSON.parse(
        File.read(@path)
      )

    validate!(data)

    data
  rescue JSON::ParserError,
         KeyError,
         TypeError,
         ArgumentError
    delete_unlocked
    nil
  end
end

def reserve_counter(
  private_key:,
  session:
)
  with_lock do
    data =
      load_unlocked

    unless data
      raise Tesla::ProtocolError,
            "session cache is empty"
    end

    cached_session =
      Tesla::Session.from_h(
        data,
        private_key: private_key
      )

    unless Crypto.secure_compare(
      cached_session.epoch,
      session.epoch
    )
      raise Tesla::ProtocolError,
            "cached session epoch changed"
    end

    unless Crypto.secure_compare(
      cached_session.routing_address,
      session.routing_address
    )
      raise Tesla::ProtocolError,
            "cached session routing address changed"
    end

    if cached_session.current_counter >
       session.current_counter
      session =
        cached_session
    end

    session.advance_counter

    save_unlocked(
      session.to_h
    )

    session
  end
end

      data

  end

  true

def save_unlocked(data)
  directory =
    File.dirname(@path)

  FileUtils.mkdir_p(
    directory
  )

  temp_path = nil

  begin
    Tempfile.create(
      [".session-", ".tmp"],
      directory
    ) do |file|
      temp_path = file.path

      File.chmod(
        FILE_MODE,
        temp_path
      )

      file.write(
        JSON.generate(data)
      )

      file.flush
      file.fsync

      File.rename(
        temp_path,
        @path
      )

      temp_path = nil
    end
  ensure
    if temp_path &&
       File.exist?(temp_path)
      File.unlink(
        temp_path
      )
    end
  end

  File.chmod(
    FILE_MODE,
    @path
  )
end
