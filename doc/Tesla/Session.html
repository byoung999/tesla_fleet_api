Overview

In-memory representation of a Tesla signed-command session, including the derived session key, vehicle public key, epoch, counter, vehicle clock offset, and routing address. The class enforces monotonic clock and counter updates and validates persisted state before restoring it.

def self.from_h(
  data,
  private_key:
)
  session_key =
    decode_hex_field(
      data,
      "session_key",
      16
    )

  vehicle_public_key_bytes =
    decode_hex_field(
      data,
      "vehicle_public_key",
      65
    )

  epoch =
    decode_hex_field(
      data,
      "epoch",
      16
    )

  routing_address =
    decode_hex_field(
      data,
      "routing_address",
      16
    )

  vehicle_public_key =
    Crypto.public_key_from_bytes(
      vehicle_public_key_bytes
    )

  expected_session_key =
    Crypto.session_key(
      private_key,
      vehicle_public_key
    )

  unless Crypto.secure_compare(
    expected_session_key,
    session_key
  )
    raise ArgumentError,
          "cached session key does not match private key"
  end

  new(
    private_key: private_key,
    vehicle_public_key: vehicle_public_key,
    session_key: session_key,
    epoch: epoch,
    counter: Integer(data.fetch("counter")),
    clock_time: Integer(data.fetch("clock_time")),
    clock_offset: Integer(data.fetch("clock_offset")),
    routing_address: routing_address
  )
end

def to_h
  {
    "session_key" =>
      @session_key.unpack1("H*"),

    "vehicle_public_key" =>
      Crypto.encoded_public_key(
        @vehicle_public_key
      ).unpack1("H*"),

    "epoch" =>
      @epoch.unpack1("H*"),

    "counter" =>
      @counter,

    "clock_time" =>
      @clock_time,

    "clock_offset" =>
      @clock_offset,

    "routing_address" =>
      @routing_address.unpack1("H*")
  }
end

def update_clock(
  clock_time,
  local_time: Time.now.to_i
)
  if clock_time < @clock_time
    raise Tesla::ProtocolError,
          "vehicle clock regressed"
  end

  @clock_time = clock_time
  @clock_offset = clock_time - local_time
end
