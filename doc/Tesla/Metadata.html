Overview

Encodes the compact tag/length/value metadata section used inside Tesla RoutableMessages. Values are stored in tag order and serialized with a 0xff terminator.

def put(tag, value)
  unless Integer === tag && tag.between?(0, 254)
    raise ArgumentError,
          "metadata tag must be between 0 and 254"
  end

  @values[tag] = encode_value(value)

  self
end

def put_byte(tag, value)
  unless Integer === value && value.between?(0, 255)
    raise ArgumentError,
          "metadata byte value out of range"
  end

  put(tag, [value].pack("C"))
end

def serialize
  result = +"".b

  @values
    .sort_by { |tag, _| tag }
    .each do |tag, value|

    result << [tag].pack("C")
    result << [value.bytesize].pack("C")
    result << value
  end

  result << [TAG_END].pack("C")

  result
end
