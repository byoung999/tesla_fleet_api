Overview

Maps signed-command exceptions to retry-policy categories. Only transport failures are classified as retryable; authentication, TLS, timeout, protocol, and unknown failures are deliberately not retried.

def classify(error)
  case error
  when Tesla::AuthenticationError
    :authentication

  when Tesla::TLSConnectionError
    :tls

  when Tesla::TimeoutError
    :timeout

  when Tesla::ProtocolError
    :protocol

  when Tesla::TransportError
    :transport

  else
    :unknown
  end
end

def retryable?(error)
  case classify(error)
  when :transport
    true
  else
    false
  end
